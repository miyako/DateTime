cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(datetime LANGUAGES CXX)

set(ICU_ROOT "" CACHE PATH "Pinned static ICU installation produced by scripts/build_icu.py")
set(DATETIME_ICU_ROOT "" CACHE PATH "Alias for ICU_ROOT used by the repository orchestrator")
if(DATETIME_ICU_ROOT)
  if(ICU_ROOT AND NOT ICU_ROOT STREQUAL DATETIME_ICU_ROOT)
    message(FATAL_ERROR "ICU_ROOT and DATETIME_ICU_ROOT disagree")
  endif()
  set(ICU_ROOT "${DATETIME_ICU_ROOT}")
endif()
if(NOT EXISTS "${ICU_ROOT}/icu-build.json")
  message(FATAL_ERROR "Build the pinned source ICU first using scripts/build_icu.py; ICU_ROOT must identify its installation")
endif()
if(WIN32)
  set(ICU_LIBS "${ICU_ROOT}/lib/icuin.lib" "${ICU_ROOT}/lib/icuuc.lib" "${ICU_ROOT}/lib/icudt.lib")
else()
  set(ICU_LIBS "${ICU_ROOT}/lib/libicui18n.a" "${ICU_ROOT}/lib/libicuuc.a" "${ICU_ROOT}/lib/libicudata.a")
endif()
foreach(library IN LISTS ICU_LIBS)
  if(NOT EXISTS "${library}")
    message(FATAL_ERROR "Missing static ICU library: ${library}")
  endif()
endforeach()
add_executable(datetime main.cpp)
add_executable(datetime-invalid-data main.cpp test_invalid_data.cpp)
list(GET ICU_LIBS 0 ICU_I18N)
list(GET ICU_LIBS 1 ICU_COMMON)
target_link_libraries(datetime PRIVATE ${ICU_LIBS})
target_link_libraries(datetime-invalid-data PRIVATE "${ICU_I18N}" "${ICU_COMMON}")
foreach(target datetime datetime-invalid-data)
  target_include_directories(${target} PRIVATE "${ICU_ROOT}/include")
  target_compile_definitions(${target} PRIVATE U_STATIC_IMPLEMENTATION)
  target_compile_features(${target} PRIVATE cxx_std_17)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /O1 /Gy /Gw)
    target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF)
    target_link_libraries(${target} PRIVATE advapi32)
  elseif(APPLE)
    target_compile_options(${target} PRIVATE -Oz -ffunction-sections -fdata-sections -fvisibility=hidden)
    target_link_options(${target} PRIVATE -Wl,-dead_strip -Wl,-no_exported_symbols)
  else()
    message(FATAL_ERROR "Only native macOS and Windows MSVC builds are supported")
  endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME datetime-native COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_datetime_native.py" --executable "$<TARGET_FILE:datetime>"
  --invalid-data-executable "$<TARGET_FILE:datetime-invalid-data>")
